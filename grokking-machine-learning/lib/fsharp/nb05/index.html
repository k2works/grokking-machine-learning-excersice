    #05 ロジスティック回帰の学習（F#）¶

    原著ノートブック: Chapter_06_Logistic_Regression/Coding_logistic_regression.ipynb

    #04 と同じ形の 8 点を、今度はロジスティック回帰で分けます。 ステップ関数がシグモイドに、パーセプトロン誤差が対数損失に置き換わります。

    ML.NET には LbfgsLogisticRegression があり、 scikit-learn が既定で使う lbfgs ソルバと 名前まで一致 します。 3 言語のなかでいちばん素直に対応が付く回でした。

    サンプル実装¶

    項目 内容

    テスト apps/grokking-ml-fsharp-lib/tests/GrokkingMlLib.Tests/Nb05LogisticRegressionTests.fs
    実行 dotnet test

    境界の型¶

    #04 の Boundary を、対数損失を持つ形に拡張します。

    /// 分離直線。重みとバイアスの組
type Boundary =
    { Weights: float[]
      Bias: float }

    /// 重み付き和にバイアスを足したもの
    member this.Score(features: int[]) =
        this.Bias
        + (Array.map2 (fun w (f: int) -> w * float f) this.Weights features |> Array.sum)

    /// 予測確率。0 / 1 ではなく 0〜1 の連続値を返す
    member this.Prediction(features: int[]) = sigmoid (this.Score features)

    /// 対数損失。当たっていても 0 にはならず、確信の度合いで連続的に変わる
    member this.LogLoss(features: int[], label: int) =
        let pred = this.Prediction features
        -float label * log pred - float (1 - label) * log (1.0 - pred)

    -float label * log pred の float 変換が 2 か所要ります。 Kotlin は Int * Double を暗黙に広げますが、F# は広げません。

    面倒に見えますが、この厳しさが効く場面もあります。 (1 - label) を float に変換し忘れると型エラーで止まるので、 整数除算のような静かな間違いが起きません。

    原著の「別の書き方」は対数損失と一致しない¶

    Python 版 で導出したとおり、原著の式は誤りです。 pred は 0〜1 の確率なので (pred - label) が ±1 になりません。

    /// 原著が「対数損失の別の書き方」として示す式。
///
/// **実際には対数損失と一致しない。** `pred` は 0〜1 の確率なので、
/// `(pred - label)` は -1 か +1 ではなく中間の値になる。
let alternateLogLossOriginal (boundary: Boundary) (features: int[]) (label: int) =
    softRelu ((boundary.Prediction features - float label) * boundary.Score features)

/// 対数損失と厳密に等しい「別の書き方」。
///
/// ラベルが 0 なら +1、1 なら -1 を掛ける。つまり `(1 - 2 * label)`。
let alternateLogLoss (boundary: Boundary) (features: int[]) (label: int) =
    softRelu (float (1 - 2 * label) * boundary.Score features)

    2 つの式を並べると、誤りの箇所が一目で分かります。 片方は boundary.Prediction features - float label、 もう片方は float (1 - 2 * label)。 前者は確率、後者は符号です。まったく違うものを掛けていたことが、 F# の明示的な型変換によって視覚的に区別されます。

    テストは全点・全ラベルの組み合わせで回します。

    [<Fact>]
let ``正しい別の書き方は対数損失と厳密に一致する`` () =
    // ラベルが 0 なら +1、1 なら -1 を掛ける。つまり (1 - 2 * label)
    for f in features do
        for label in [ 0; 1 ] do
            Assert.Equal(start.LogLoss(f, label), alternateLogLoss start f label, 12)

    一致しないほうは Array.iteri2 で添字つきに回し、 失敗時にどの点かが分かるメッセージを付けました。

    [<Fact>]
let ``原著の別の書き方は対数損失と一致しない`` () =
    // pred は 0〜1 の確率なので (pred - label) は ±1 にならない
    Array.iteri2
        (fun i f l ->
            if abs (start.Score f) > 1e-9 then
                Assert.True(
                    abs (alternateLogLossOriginal start f l - start.LogLoss(f, l)) > 1e-6,
                    $"index {i} で一致してしまった"
                ))
        features
        labels

    Array.iteri2 は「2 本の配列を添字つきで同時に回す」関数です。 Python の enumerate(zip(...)) にあたるものが 1 語で書けます。

    ロジスティックトリック¶

    let logisticTrick (learningRate: float) (boundary: Boundary) (features: int[]) (label: int) =
    let pred = boundary.Prediction features
    let delta = float label - pred

    { Weights =
        Array.map2 (fun w (f: int) -> w + delta * float f * learningRate) boundary.Weights features
      Bias = boundary.Bias + delta * learningRate }

    #04 では、原著のバグを再現するために * float features.Length を掛ける必要がありました。ここでは不要です。 原著もこの章ではバイアスの更新をループの外に書いています。

    delta を先に束縛しているのは、重みとバイアスの両方で使うからです。 F# のレコード式は各フィールドが独立した式なので、 共通の中間値は外に出さないと 2 回計算することになります。

    ML.NET の LbfgsLogisticRegression¶

    /// 同じ問題を ML.NET の `LbfgsLogisticRegression` に解かせる。
///
/// scikit-learn の `LogisticRegression` は既定で lbfgs ソルバを使う。
/// ML.NET も同名のトレーナーを持つので、いちばん近い対応になる。
let fitWithMlNet (features: int[][]) (labels: int[]) =
    let ctx = MLContext(seed = 0)

    let rows =
        Array.map2
            (fun (f: int[]) (l: int) ->
                { Features = f |> Array.map single
                  Label = (l = 1) })
            features
            labels

    let data = ctx.Data.LoadFromEnumerable rows
    let model = ctx.BinaryClassification.Trainers.LbfgsLogisticRegression().Fit data
    let engine = ctx.Model.CreatePredictionEngine<Sample, Prediction> model

    fun (row: int[]) ->
        let prediction =
            engine.Predict { Features = row |> Array.map single; Label = false }

        if prediction.PredictedLabel then 1 else 0

    #04 と同じ作法です。[<CLIMutable>] のレコード、float32 の特徴量、 bool のラベル、PredictionEngine で 1 行ずつ予測。 型を private にすると実行時に TypeLoadException になる点も同じなので、 一度踏んだ落とし穴は繰り返しませんでした。

    3 言語のトレーナー対応¶

    言語 ライブラリのクラス ソルバ 既定の正則化
    Python sklearn.linear_model.LogisticRegression lbfgs あり（L2, C=1.0）
    Kotlin smile.classification.LogisticRegression.binomial 準ニュートン なし（lambda=0）
    F# ctx.BinaryClassification.Trainers.LbfgsLogisticRegression() lbfgs あり（L1 + L2）

    名前が同じでも既定が違います。Smile だけ正則化なしです。 ML.NET は L1 と L2 の両方に既定値を持っており、 scikit-learn の L2 のみとも厳密には違います。

    「ロジスティック回帰を使った」だけでは、解いた問題が特定できません。 係数を言語間で突き合わせるのが無意味なのはこのためで、 テストは 3 言語とも予測結果で書いています。

    [<Fact>]
let ``ML_NET のロジスティック回帰も全点を正しく分類する`` () =
    // 原著 scikit-learn の出力: Logistic Regression Predictions: [0 0 0 0 1 1 1 1]
    let predict = fitWithMlNet features labels

    Assert.Equal<int[]>(labels, features |> Array.map predict)

    3 言語の収束比較¶

    種を固定した手書きループで、全点が正しく分類できるまでのエポック数です。

    実装 500 エポック 5,000 エポック 20,000 エポック
    Python 1 点誤り 全点正解 全点正解
    Kotlin 2 点誤り 1 点誤り 全点正解
    F# 2 点誤り 全点正解 全点正解

    F# の 5,000 エポック時点の重みは [1.706, 1.056] / バイアス -3.999。 Python の [1.722, 1.075] / -3.999 とよく似た位置に来ています。 乱数列は別なのに、到達点はほぼ同じ です。

    一方 Kotlin は同じ 5,000 エポックで [1.723, 1.149] / -3.966 と、 数値は近いのに 1 点だけ判定が反転しています。 境界のすぐ上に点があるので、わずかな差が予測を分けているのです。

    このデータには、境界線上ぎりぎりの点があります。 #04 のパーセプトロンは 500 エポックで揃ったので、 対数損失がこの点をどれだけゆっくり押し切るか が言語ごとの差になりました。

    この章で F# が効いた場面・効かなかった場面¶

    場面 評価
    2 つの式の対比 効いた — 明示的な型変換で「確率」と「符号」の違いが見える
    Array.iteri2 効いた — 2 本の配列を添字つきで回すのが 1 語
    ロジスティックトリック 効いた — #04 と違い原著が正しいので素直に書ける
    ML.NET のトレーナー名 効いた — scikit-learn と同じ lbfgs で対応が明確
    型変換 引き分け — float 変換が要るが、静かな間違いを防ぐ
    float32 への変換 効かなかった — Array.map single を毎回書く
    既定の正則化 効かなかった — L1 + L2 で scikit-learn と厳密には違う
